% Screenshot with normalized [0,1]x[0,1] TikZ overlay annotations
% Origin (0,0) is bottom-left of the image, (1,1) is top-right.
% Usage: \guideannotated[width-spec]{filename}{Caption text}{fig:label}{\highlightbox{...}\stepbadge{...}}
\newcommand{\guideannotated}[5][0.94\linewidth]{%
  \begin{figure}[H]
    \centering
    \fcolorbox{FarmBorder}{white}{%
      \begin{tikzpicture}
        \node[anchor=south west,inner sep=0] (img) at (0,0) {%
          \includegraphics[width=#1,height=0.58\textheight,keepaspectratio]{#2}%
        };
        \begin{scope}[x={(img.south east)},y={(img.north west)}]
          #5
        \end{scope}
      \end{tikzpicture}%
    }%
    \caption{#3}%
    \label{#4}%
  \end{figure}%
}

% Annotation helpers for use inside \guideannotated
\newcommand{\highlightbox}[4]{%
  \draw[GuideRed,line width=1.6pt,rounded corners=2pt] (#1,#2) rectangle (#3,#4);%
}
\newcommand{\stepbadge}[3]{%
  \node[circle,fill=GuideRed,text=white,font=\sffamily\bfseries\scriptsize,inner sep=1.2pt,minimum size=13pt] at (#1,#2) {#3};%
}
